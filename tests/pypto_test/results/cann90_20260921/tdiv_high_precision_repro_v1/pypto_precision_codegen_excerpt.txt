382: // the exact PTOAS enum attribute after outs(...) for high-precision mode.
383: // Unlike cmp/cvt attributes, precision-op assembly formats place their
384: // attr-dict after the complete ins()/outs() clause.
385: static std::string MakePrecisionCodegenPTO(const std::string& pto_op_name, size_t arity,
386:                                            const char* attr_kind, const CallPtr& op,
387:                                            codegen::CodegenBase& codegen_base) {
388:   auto& codegen = AsPto(codegen_base);
389:   CheckArity(op, pto_op_name, arity);
390:   const bool high_precision = op->GetKwarg<bool>("high_precision", false);
391:   std::string code = pto_op_name + " " + GenerateInsOutsClause(op, codegen);
392:   if (high_precision) {
393:     code += " {precisionType = #pto<";
394:     code += attr_kind;
395:     code += " high_precision>}";
396:   }
397:   codegen.Emit(code);
398:   return "";
399: }
