341:         RegTensor<T>& reg_dst, RegTensor<T>& reg_src0, RegTensor<T>& reg_src1, MaskReg& preg)
342:     {

344:             DivDiffCompensationFloatImpl<T, RegTensor<T> >(reg_dst, reg_src0, reg_src1, preg);

346:             DivIEEE754HalfImpl<T, RegTensor<T> >(reg_dst, reg_src0, reg_src1, preg);
347:         } else {
348:             vdiv(reg_dst, reg_src0, reg_src1, preg, MODE_ZEROING);
349:         }
350:     }
