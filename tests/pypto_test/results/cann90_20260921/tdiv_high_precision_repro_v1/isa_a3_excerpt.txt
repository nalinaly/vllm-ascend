20: template <typename T>
21: struct DivOp {
22:     PTO_INTERNAL static void BinInstr(__ubuf__ T* dst, __ubuf__ T* src0, __ubuf__ T* src1, uint8_t repeats)
23:     {
24:         vdiv(dst, src0, src1, repeats, 1, 1, 1, 8, 8, 8);
25:     }
26:     PTO_INTERNAL static void BinInstr(
27:         __ubuf__ T* dst, __ubuf__ T* src0, __ubuf__ T* src1, uint8_t repeats, uint8_t dstRepeatStride,
28:         uint8_t src0RepeatStride, uint8_t src1RepeatStride)
29:     {
30:         vdiv(dst, src0, src1, repeats, 1, 1, 1, dstRepeatStride, src0RepeatStride, src1RepeatStride);
31:     }
32: };
33: 
34: template <
35:     typename TileData, unsigned elementsPerRepeat, unsigned blockSizeElem, unsigned dstRowStride,
36:     unsigned src0RowStride = dstRowStride, unsigned src1RowStride = dstRowStride>
37: __tf__ PTO_INTERNAL void TDiv(
38:     typename TileData::TileDType __out__ dst, typename TileData::TileDType __in__ src0,
39:     typename TileData::TileDType __in__ src1, unsigned validRows, unsigned validCols)
40: {
41:     using T = typename TileData::DType;
42:     __ubuf__ T* dstPtr = (__ubuf__ T*)__cce_get_tile_ptr(dst);
43:     __ubuf__ T* src0Ptr = (__ubuf__ T*)__cce_get_tile_ptr(src0);
44:     __ubuf__ T* src1Ptr = (__ubuf__ T*)__cce_get_tile_ptr(src1);
45:     if constexpr (dstRowStride == src0RowStride && dstRowStride == src1RowStride) {
46:         BinaryInstr<DivOp<T>, T, TileData, elementsPerRepeat, blockSizeElem, dstRowStride>(
47:             dstPtr, src0Ptr, src1Ptr, validRows, validCols);
48:     } else {
49:         BinaryInstr<DivOp<T>, T, elementsPerRepeat, blockSizeElem, dstRowStride, src0RowStride, src1RowStride>(
50:             dstPtr, src0Ptr, src1Ptr, validRows, validCols);
51:     }
52:     return;
53: }

55: template <typename T, typename TileDataDst, typename TileDataSrc0, typename TileDataSrc1>
56: PTO_INTERNAL void TDivCheck(const TileDataDst& dst, const TileDataSrc0& src0, const TileDataSrc1& src1)
57: {
58:     static_assert(
59:         std::is_same<T, half>::value || std::is_same<T, float16_t>::value || std::is_same<T, float>::value ||
60:             std::is_same<T, float32_t>::value,
61:         "Fix: TDIV has invalid data type.");
62:     static_assert(
63:         TileDataDst::isRowMajor && TileDataSrc0::isRowMajor && TileDataSrc1::isRowMajor,
64:         "Fix: TDIV only support row major layout.");
65:     unsigned validRows = dst.GetValidRow();
66:     unsigned validCols = dst.GetValidCol();
67:     PTO_ASSERT(
68:         src0.GetValidRow() == validRows && src0.GetValidCol() == validCols,
69:         "Fix: TDIV input tile src0 valid shape mismatch with output tile dst shape.");
70:     PTO_ASSERT(
71:         src1.GetValidRow() == validRows && src1.GetValidCol() == validCols,
72:         "Fix: TDIV input tile src1 valid shape mismatch with output tile dst shape.");
73: }

75: template <
76:     auto PrecisionType = DivAlgorithm::DEFAULT, typename TileDataDst, typename TileDataSrc0, typename TileDataSrc1>
77: PTO_INTERNAL void TDIV_IMPL(TileDataDst& dst, TileDataSrc0& src0, TileDataSrc1& src1)
78: {
79:     using T = typename TileDataDst::DType;
80:     TDivCheck<T, TileDataDst, TileDataSrc0, TileDataSrc1>(dst, src0, src1);
81:     constexpr unsigned blockSizeElem = BLOCK_BYTE_SIZE / sizeof(T);
82:     constexpr unsigned elementsPerRepeat = REPEAT_BYTE / sizeof(T);
83:     // when tileshape of src0, src1 and dst are the same, validRows and validCols are also the same
84:     if constexpr (std::is_same_v<TileDataDst, TileDataSrc0> && std::is_same_v<TileDataDst, TileDataSrc1>) {
85:         constexpr unsigned dstRowStride = TileDataDst::RowStride;
86:         TDiv<TileDataDst, elementsPerRepeat, blockSizeElem, dstRowStride>(
87:             dst.data(), src0.data(), src1.data(), dst.GetValidRow(), dst.GetValidCol());
88:     } else {
89:         // when tileshape of src0, src1 and dst are different, validRows and validCols are also the same
90:         constexpr unsigned dstRowStride = TileDataDst::RowStride;
91:         constexpr unsigned src0RowStride = TileDataSrc0::RowStride;
92:         constexpr unsigned src1RowStride = TileDataSrc1::RowStride;
93:         TDiv<TileDataDst, elementsPerRepeat, blockSizeElem, dstRowStride, src0RowStride, src1RowStride>(
94:             dst.data(), src0.data(), src1.data(), dst.GetValidRow(), dst.GetValidCol());
95:     }
96: }
