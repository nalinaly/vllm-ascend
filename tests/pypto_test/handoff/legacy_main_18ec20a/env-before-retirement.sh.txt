#!/usr/bin/env bash
# Source this file to use the local CANN 9.0 PTO development environment.
export PTO_EAGER_ROOT="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
source /usr/local/Ascend/cann-9.0.0/set_env.sh
export ASCEND_HOME_PATH=/usr/local/Ascend/cann-9.0.0
export GCC15_ROOT=/data/ci-runner/pypto-toolchain/toolchain/2026.08.5/gcc
export PTOAS_ROOT="$PTO_EAGER_ROOT/vllm-ascend-dsv4-pto/.cache/csa/ptoas-0.61"
export PATH="$PTOAS_ROOT/bin:$GCC15_ROOT/bin:$PATH"
export CC="$GCC15_ROOT/bin/gcc-15"
export CXX="$GCC15_ROOT/bin/g++-15"
export LD_LIBRARY_PATH="$GCC15_ROOT/lib64${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}"
export PYTHONNOUSERSITE=1
source "$PTO_EAGER_ROOT/.venv/bin/activate"
if [[ -f "$PTO_EAGER_ROOT/vllm-ascend-dsv4-pto/.cache/csa/atb-package-9.0.0/atb/set_env.sh" ]]; then
  # Local NNAL 9.0.0 archive; torch 2.10 in this environment uses CXX11 ABI.
  source "$PTO_EAGER_ROOT/vllm-ascend-dsv4-pto/.cache/csa/atb-package-9.0.0/atb/set_env.sh" --cxx_abi=1
fi
if [[ -f "$PTO_EAGER_ROOT/vllm-ascend-dsv4-pto/.cache/csa/csa-native-ops-install/vendors/custom_transformer/bin/set_env.bash" ]]; then
  source "$PTO_EAGER_ROOT/vllm-ascend-dsv4-pto/.cache/csa/csa-native-ops-install/vendors/custom_transformer/bin/set_env.bash"
fi
