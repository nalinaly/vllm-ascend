#!/usr/bin/env bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "${SCRIPT_DIR}/../config.sh"
cd "${SCRIPT_DIR}"

local_ip="$(detect_local_ip)"
if [[ "${local_ip}" != "${PREFILL_IP}" ]]; then
  printf 'Prefill must run on %s, got %s\n' "${PREFILL_IP}" "${local_ip}" >&2
  exit 2
fi
case "${KV_DECODE_DP_SIZE}" in
  16|32) ;;
  *) printf 'KV_DECODE_DP_SIZE must be 16 or 32\n' >&2; exit 2 ;;
esac
export KV_DECODE_DP_SIZE

mkdir -p ../logs
LOG_FILE="../logs/prefill_dp${KV_DECODE_DP_SIZE}_$(date +%Y%m%d_%H%M%S).log"

python3 ../common/launch_online_dp.py \
  --dp-size 4 \
  --tp-size 4 \
  --dp-size-local 4 \
  --dp-rank-start 0 \
  --dp-address "${local_ip}" \
  --dp-rpc-port "${DP_RPC_PORT}" \
  --vllm-start-port "${VLLM_START_PORT}" \
  2>&1 | tee -a "${LOG_FILE}"
