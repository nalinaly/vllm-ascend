#!/usr/bin/env bash
# Isolated toolchain for pto_eager (pypto + simpler kernel-mode-integration-test).
# Do not source inductor/env.sh — that points at grok_pypto.

export PTO_EAGER_ROOT="/mnt/workspace/inductor/pto_eager"
export CANN_ROOT="${CANN_ROOT:-/home/developer/Ascend/cann-9.2.0}"
if [[ -f "$CANN_ROOT/set_env.sh" ]]; then
    # shellcheck disable=SC1091
    source "$CANN_ROOT/set_env.sh"
fi

export LLVM_BUILD_DIR="/mnt/workspace/llvm-project-vpto-src/build-release-cann9.2-clean"
export PTOAS_INSTALL_ROOT="/mnt/workspace/inductor/pto/PTOAS/install-v0.57-llvm21-cann9.2-clean"
export PTOAS_ROOT="$PTOAS_INSTALL_ROOT/bin"
export GCC15_ROOT="/mnt/workspace/inductor/toolchains/gcc15"
export GCC15_SHIMS="/mnt/workspace/inductor/toolchains/gcc15-shims"
export CCACHE_DIR="/mnt/workspace/inductor/.ccache"

_pto_eager_prepend() {
    local variable="$1"
    local directory="$2"
    local current="${!variable-}"
    [[ -d "$directory" ]] || return 0
    case ":$current:" in
        *":$directory:"*) ;;
        *) export "$variable=$directory${current:+:$current}" ;;
    esac
}

_pto_eager_prepend PATH "$PTOAS_ROOT"
_pto_eager_prepend PATH "$GCC15_SHIMS"
_pto_eager_prepend PATH "$CANN_ROOT/tools/ccec_compiler/bin"
_pto_eager_prepend LD_LIBRARY_PATH "$LLVM_BUILD_DIR/lib"
_pto_eager_prepend LD_LIBRARY_PATH "$PTOAS_INSTALL_ROOT/lib"
_pto_eager_prepend LD_LIBRARY_PATH "$GCC15_ROOT/lib"
# pypto._torch_npu is linked against libtorch_npu.so; RPATH does not reach user-site.
_pto_eager_prepend LD_LIBRARY_PATH "$HOME/.local/lib/python3.11/site-packages/torch_npu/lib"
# vLLM-Ascend registers ATB extensions during worker startup.  The torch_atb
# wheel keeps libatb and its companion libraries in a sibling user-site tree.
_pto_eager_prepend LD_LIBRARY_PATH "$HOME/.local/lib/python3.11/site-packages/torch_atb/lib"
# msprof_analysis.so needs unversioned libsqlite3.so (CANN timeline / kernel export).
_pto_eager_prepend LD_LIBRARY_PATH "/usr/local/sqlite3/lib"
_pto_eager_prepend LD_LIBRARY_PATH "$CANN_ROOT/tools/profiler/profiler_tool/analysis/lib64"
_pto_eager_prepend PYTHONPATH "$CANN_ROOT/tools/profiler/profiler_tool/analysis/lib64"

# Avoid picking up /mnt/workspace/inductor/triton as a namespace package when cwd
# is the inductor root; that breaks torch_npu (triton.language missing).
export TORCH_DEVICE_BACKEND_AUTOLOAD="${TORCH_DEVICE_BACKEND_AUTOLOAD:-0}"

# CANN set_env.sh and the workspace injector both prepend PYTHONPATH.
# Keep pto_eager first so this tree wins over cann's pypto 0.2.1 and
# the user-site simpler editable hook.
export PYTHONPATH="${PTO_EAGER_ROOT}/simpler:${PTO_EAGER_ROOT}/simpler/python:${CANN_ROOT}/python/site-packages:${CANN_ROOT}/opp/built-in/op_impl/ai_core/tbe:${CANN_ROOT}/tools/profiler/profiler_tool/analysis/lib64"

if [[ -f "$PTO_EAGER_ROOT/.venv/bin/activate" ]]; then
    # shellcheck disable=SC1091
    source "$PTO_EAGER_ROOT/.venv/bin/activate"
fi

unset -f _pto_eager_prepend
